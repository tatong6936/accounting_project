3 สร้าง Schema & Service
1
สร้าง db/schema.cds
namespace accounting;
using { cuid } from '@sap/cds/common';

entity Batches : cuid {
  hisDocumentType         : String(30);   // INVOICE_OPD, RECEIPT_OPD, ...
  hospitalCode            : String(20);
  accountingDocumentNumber: String(50);
  billDate                : Date;
  currency                : String(3);
  hospitalNumber          : String(50);
  visitNumber             : String(50);
  coverageType            : String(50);
  patientGroup            : String(50);
  marketChannel           : String(50);
  items                   : Composition of many Items on items.batch = $self;
  payments                : Composition of many Payments on payments.batch = $self;
  journalEntryPostings    : Composition of many JournalEntryPostings on journalEntryPostings.batch = $self;
}

entity Items : cuid {
  batch                   : Association to Batches;
  itemCode                : String(50);
  itemName                : String(200);
  totalAmount             : Decimal(15,2);
  discountAmount          : Decimal(15,2);
  netAmount               : Decimal(15,2);
  billingGroup            : String(50);
  requestingDepartmentCode: String(50);
  performingDepartmentCode: String(50);
  patientType             : String(50);
  orderCategory           : String(50);
  orderSubCategory        : String(50);
  fiPosting               : Association to JournalEntryPostings;
}

entity Payments : cuid {
  batch        : Association to Batches;
  paymentMethod: String(20);  // CASH, TRANSFER, EDC, QR, VOUCHER, DEPOSIT
  paymentAmount: Decimal(15,2);
  bankCode     : String(20);
  edcTerminalId: String(50);
}

entity JournalEntryPostings : cuid {
  batch                   : Association to Batches;
  JournalEntryDocumentNumber: String(20);
  JournalEntryCompanyCode : String(10);
  JournalEntryYear        : String(4);
  JournalEntryStatus      : String(10);  // POSTED / ERROR
  messages                : Composition of many JournalEntryMessages on messages.journalEntryPosting = $self;
}

entity JournalEntryMessages : cuid {
  journalEntryPosting: Association to JournalEntryPostings;
  lineNumber         : Integer;
  message            : String(500);
  msgType            : String(1);  // S=Success, I=Info, E=Error, W=Warning
}
2
สร้าง srv/accounting-service.cds
using { accounting } from '../db/schema';

service AccountingService @(path: '/accounting', requires: 'authenticated-user') {

  entity Batches            as projection on accounting.Batches;
  entity Items              as projection on accounting.Items;
  entity Payments           as projection on accounting.Payments;
  entity JournalEntryPostings as projection on accounting.JournalEntryPostings;
  entity JournalEntryMessages as projection on accounting.JournalEntryMessages;

  // Action: รับข้อมูลจาก HIS
  action import(batches: array of {
    hisDocumentType: String; hospitalCode: String;
    accountingDocumentNumber: String; billDate: Date; currency: String;
    hospitalNumber: String; visitNumber: String; coverageType: String;
    patientGroup: String; marketChannel: String;
    items: array of {}; payments: array of {};
  }) returns array of String;

  // Action: ส่ง FI Document ไป S/4HANA
  action postFI(
    batchId: String;
    hisDocumentType: String;
    billDate: Date;
  ) returns {};
}

4 สร้าง mta.yaml

ไฟล์นี้กำหนดโครงสร้าง MTA (Multi-Target Application) สำหรับ deploy บน BTP Cloud Foundry

_schema-version: "3.3.0"
ID: accounting
version: 1.0.0

modules:
  - name: accounting-srv
    type: nodejs
    path: gen/srv             # path หลัง cds build
    parameters:
      buildpack: nodejs_buildpack
      memory: 512M
      disk-quota: 1024M
    requires:
      - name: accounting-auth
      - name: accounting-postgres
      - name: accounting-destination
      - name: central-config-auth-binding
    provides:
      - name: accounting-srv-api
        properties:
          srv-url: ${default-url}

  - name: accounting-db-deployer
    type: hdb
    path: gen/db              # deploy schema ขึ้น PostgreSQL
    requires:
      - name: accounting-postgres

resources:
  - name: accounting-auth
    type: org.cloudfoundry.managed-service
    parameters:
      service: xsuaa
      service-plan: application
      path: ./xs-security.json

  - name: accounting-postgres
    type: org.cloudfoundry.existing-service  # ใช้ existing service!
    parameters:
      service-name: my-cap-postgres-db       # ชื่อ service ที่มีอยู่แล้ว

  - name: accounting-destination
    type: org.cloudfoundry.managed-service
    parameters:
      service: destination
      service-plan: lite

  - name: central-config-auth-binding
    type: org.cloudfoundry.existing-service  # ใช้ existing service!
    parameters:
      service-name: central-config-auth      # shared XSUAA จาก central-config
⚡ สำคัญ: accounting-postgres และ central-config-auth-binding ใช้ type org.cloudfoundry.existing-service — ไม่ใช่ managed-service ดังนั้น service เหล่านี้ต้องมีอยู่ใน CF Space ก่อน

6 BTP Services ที่ต้องสร้าง
Service Name (ใน CF) Type Plan Action หมายเหตุ
my-cap-postgres-db PostgreSQL on BTP free / standard EXISTING ต้องมีอยู่ก่อน — ใช้ร่วมกับ project อื่น
central-config-auth XSUAA application EXISTING ต้องมีอยู่ก่อน — เป็น XSUAA ของ central-config-srv

7 Local Dev Setup
⚡ Local dev ต้อง connect กับ PostgreSQL ใน BTP โดยตรงผ่าน SSH tunnel — ไม่ใช่ local DB
1
เปิด SSH Tunnel ไป PostgreSQL

เปิด terminal แยก และเปิดค้างไว้ตลอดระหว่าง dev:

# เปิด tunnel: local port 15432 → BTP PostgreSQL port 5887
cf ssh accounting-srv -L 15432:<pg-host>:5887

# <pg-host> ดูจาก cf env accounting-srv → VCAP_SERVICES → postgresql-db → host
# ถ้า accounting-srv ยังไม่ได้ deploy ให้ใช้ service ที่ deploy แล้วตัวอื่น
# เช่น cf ssh central-config-srv -L 15432:<pg-host>:5887
2
สร้าง .cdsrc-private.json

ไฟล์นี้บอก CAP ให้ connect PostgreSQL ผ่าน SSH tunnel:

{
  "requires": {
    "db": {
      "kind": "postgres",
      "credentials": {
        "host": "localhost",
        "port": 15432,
        "database": "<database-name>",
        "user": "<username>",
        "password": "<password>",
        "schema": "accounting"
      }
    }
  }
}

// ดึงค่าจาก: cf env accounting-srv → VCAP_SERVICES → postgresql-db → credentials

11 Error Troubleshooting
Error Message สาเหตุ วิธีแก้
Tax statement item missing for tax code O0 ProductTaxItem ไม่ได้ถูกส่งใน SOAP body หรือ TaxItemClassification ไม่ถูก ตรวจสอบ buildRequestBody ว่ามี <ProductTaxItem> พร้อม TaxCode=O0 และ TaxItemClassification=MWS
KSCHL is empty SAP ยังไม่ได้ config ConditionType สำหรับ tax code O0 ในระบบ S/4HANA ให้ SAP Basis/Config admin เข้าไป config customizing ใน S/4HANA — ไม่ใช่ปัญหาของ code
G/L account requires valid tax code GL account ที่ใช้เป็น tax-relevant แต่ไม่ได้ส่ง TaxCode ในบาง item line ตรวจสอบ buildItemLines ว่าทุก line ที่ใช้ GL นั้นมี <TaxCode> ด้วย
central-config credentials not found Env vars สำหรับ XSUAA ไม่ได้ถูก set ในเทอร์มินัลที่รัน cds serve ตรวจสอบ .env ว่ามีค่าครบ แล้ว source .env หรือรัน cds serve ใน terminal เดียวกัน
Connection to localhost:15432 refused SSH tunnel ไป PostgreSQL ไม่ได้เปิดอยู่ เปิด terminal แยก แล้วรัน: cf ssh accounting-srv -L 15432:<pg-host>:5887
XSUAA 401 Unauthorized XSUAA client_secret เก่าหรือหมดอายุ รัน cf env central-config-srv แล้วดึง clientsecret ใหม่ไปอัปเดตใน .env หรือ cf set-env
ECONNREFUSED PostgreSQL SSH tunnel หลุดหรือ PostgreSQL credentials ผิด ตรวจสอบ tunnel ว่ายังทำงาน + ตรวจสอบค่าใน .cdsrc-private.json ให้ตรงกับ VCAP_SERVICES
Module not found: s4h-journal.cjs ไม่พบไฟล์ หรือ path ผิดใน require() ตรวจสอบว่าไฟล์อยู่ที่ srv/s4h-journal.cjs และ import path ใน accounting-service.js ถูกต้อง
cf deploy fails: service not found existing service ที่ระบุใน mta.yaml ไม่มีใน CF Space รัน cf services เพื่อตรวจสอบ แล้วแก้ service-name ใน mta.yaml ให้ตรงกัน
cds build failed: cannot resolve module package.json ขาด dependency รัน npm install แล้วลอง cds build ใหม่
503 Service Unavailable (หลัง deploy) App crash ระหว่าง startup — มักเกิดจาก missing env หรือ service binding ผิด รัน cf logs accounting-srv --recent เพื่อดู error แล้วแก้ตาม log
Posting is blocked (SAP error) วันที่ในเอกสาร (PostingDate) ไม่อยู่ใน open posting period ของ SAP ให้ SAP Finance open posting period สำหรับเดือนนั้น หรือใช้วันที่ที่ period ยัง open อยู่
